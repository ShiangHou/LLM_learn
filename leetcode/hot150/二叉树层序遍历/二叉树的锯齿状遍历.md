# 二叉树的锯齿状遍历
给你二叉树的根节点 root ，返回其节点值的 锯齿形层序遍历 。（即先从左往右，再从右往左进行下一层遍历，以此类推，层与层之间交替进行）。

## 初步思路
锯齿状遍历就是一个是正的，一个是负的遍历就行，但是我们的顺序应该是不变的

用一个flag来表示正负，我们只需要最后搜集到ans的时候看下flag，正着搜集和反着搜集就行

```python 
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:
            return []
        flag = 0#0表示顺着搜集，1表示逆着搜集
        from collections import deque
        q = deque()
        q.append(root)
        ans = []
        while q:
            temp = []
            n = len(q)
            for _ in range(n):
                node = q.popleft()
                temp.append(node.val)
                if node.left:
                    q.append(node.left)
                if node.right:
                    q.append(node.right)
            if flag == 0:#正常搜集
                ans.append(temp)
                flag = 1
            else:#反着搜集
                ans.append(temp[::-1])
                flag = 0
        return ans
```
