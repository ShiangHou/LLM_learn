# 搜索旋转排序数组

智普一面


```python 
#思路是二分，比较mid和left，然后判断左右大的就行

def find(n,nums,target):
    l,r = 0,n-1
    while l <= r:
        mid = (l+r) // 2
        if nums[mid] == target:
            return mid
        if nums[mid] >= nums[l]:#左边一定是递增的
            if nums[l] <= target < nums[mid]:#说明在左边
                r = mid-1
            else:
                l = mid+1
        else:#右边是有序递增的
            if nums[mid] < target<= nums[r]:
                l = mid+1
            else:
                r = mid-1


    return -1

n = int(input())
nums = list(map(int,input().split()))
target = int(input())

print(find(n,nums,target))
```
